# TankKoll Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Production-ready Swedish fuel-tracking PWA (Next.js static export) deployed to GitHub Pages at /tankkoll.

**Architecture:** Statically exported Next.js App Router shell; all data client-side via zustand store backed by an async `StorageAdapter` (LocalStorage now, REST later). Pure, unit-tested calculation functions derive every figure from raw refuelings. UI is mockup-faithful (`Mockup.png`).

**Tech Stack:** Next.js 15/16 (App Router, `output:'export'`), React 19, TypeScript strict, Tailwind v4, Recharts, zustand, lucide-react, vitest. Graphics via bildgen MCP + sharp.

## Global Constraints

- UI copy in **Swedish**; numbers/dates via `Intl` locale `sv-SE`; currency default `SEK` shown as `kr`; "mil" = 10 km.
- Colors exactly: bg `#0F172A`, card `#1E293B`, primary `#2563EB`, success `#22C55E`, warning `#F59E0B`, danger `#EF4444`. Dark is default theme.

### Task 2: Domain types + calculations (TDD)

**Interfaces produced (exact):**

```ts
// types.ts
export type FuelType = 'bensin' | 'diesel' | 'etanol' | 'el-hybrid' | 'annat';
export interface Vehicle { id: string; name: string; registrationNumber?: string; fuelType: FuelType; manufacturer?: string; model?: string; year?: number; tankSize?: number; wltpConsumption?: number; imageUrl?: string; createdAt: string; }
export interface Refueling { id: string; vehicleId: string; date: string; liters: number; odometer: number; pricePerLiter?: number; notes?: string; createdAt: string; }
export interface Settings { currency: string; theme: 'dark' | 'light'; activeVehicleId?: string; }
export interface AppData { version: 1; vehicles: Vehicle[]; refuelings: Refueling[]; settings: Settings; }
export const DEFAULT_SETTINGS: Settings = { currency: 'SEK', theme: 'dark' };
export const EMPTY_DATA: AppData = { version: 1, vehicles: [], refuelings: [], settings: DEFAULT_SETTINGS };

// calculations.ts
export interface EnrichedRefueling extends Refueling { distance?: number; consumption?: number; kmPerLiter?: number; cost?: number; costPerKm?: number; costPerMil?: number; }
export function enrichRefuelings(refuelings: Refueling[]): EnrichedRefueling[]; // sort odometer asc; first entry = baseline (no distance); distance<=0 ⇒ no consumption/cost-per-km; cost only when pricePerLiter present
export interface VehicleStats { count: number; totalDistance: number; totalLiters: number; totalCost: number; avgConsumption?: number; minConsumption?: number; maxConsumption?: number; avgPricePerLiter?: number; costPerKm?: number; costPerMil?: number; }
export function computeStats(enriched: EnrichedRefueling[]): VehicleStats; // avgConsumption = totalLiters-with-distance / totalDistance × 100 (weighted); avgPricePerLiter liter-weighted; costPerKm = totalCost(with distance) / totalDistance
export function gaugeAverage(enriched: EnrichedRefueling[], n?: number): number | undefined; // mean of last n (default 5) consumption values
export interface MonthlyAgg { month: string; /* 'YYYY-MM' */ label: string; /* 'feb' sv-SE */ cost: number; distance: number; liters: number; avgConsumption?: number; avgPrice?: number; }
export function monthlyAggregates(enriched: EnrichedRefueling[], months: number, now?: Date): MonthlyAgg[]; // last `months` calendar months incl. current, zero-filled
export function wltpDiffPercent(actual: number, official: number): number; // (actual-official)/official*100, round to whole
export function currentMonthSummary(enriched: EnrichedRefueling[], now?: Date): { distance: number; cost: number; avgConsumption?: number };
export function consumptionSeries(enriched: EnrichedRefueling[], months: number, now?: Date): { date: string; consumption: number }[];
export function priceSeries(enriched: EnrichedRefueling[], months: number, now?: Date): { date: string; price: number }[];
```
